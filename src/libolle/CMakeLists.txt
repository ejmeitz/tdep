cmake_minimum_required(VERSION 3.10)
project(MyProject LANGUAGES Fortran C CXX)

# Find all required packages
find_package(HDF5 COMPONENTS Fortran REQUIRED)
find_package(MPI COMPONENTS Fortran REQUIRED)
find_package(LAPACK REQUIRED)
find_package(FFTW3 REQUIRED)

# Setup paths that match your relative path structure
set(OBJECT_PATH "../../build/libolle/")
set(MODULE_PATH "../../inc/libolle/")
set(DEST_PATH "../../lib/")

# Create necessary directories (resolved to absolute paths)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/${OBJECT_PATH})
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/${MODULE_PATH})
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/${DEST_PATH})

# Construct library paths and includes
set(blaslapackLPATH "-L${LAPACK_LIBRARY_DIRS}")
set(blaslapackLIBS "${LAPACK_LIBRARIES}")
set(incLPATHhdf "-L${HDF5_LIBRARY_DIRS}")
set(incIPATHhdf "-I${HDF5_INCLUDE_DIRS}")
set(incLIBShdf "${HDF5_LIBRARIES}")
set(incIPATHfft "-I${FFTW3_INCLUDE_DIRS}")
set(incLIBSfft "${FFTW3_LIBRARIES}")
set(incIPATHmpi "-I${MPI_Fortran_INCLUDE_PATH}")
set(incLIBSmpi "${MPI_Fortran_LIBRARIES}")

# Compiler flags
set(MODULE_FLAG "-J")
set(OPT "${CMAKE_Fortran_FLAGS}")
set(OPTLOW "${CMAKE_Fortran_FLAGS_DEBUG}")
set(FLG0 "-Wall -Wextra")  # Adjust these flags as needed
set(FLG1 "-Wall -Wextra")  # Adjust these flags as needed

# Determine the flags combined
set(LPATH "${blaslapackLPATH} ${incLPATHhdf} ${incLPATHfft} ${incLPATHmpi}")
set(IPATH "${blaslapackIPATH} ${incIPATHhdf} ${incIPATHfft} ${incIPATHmpi}")
set(LIBS "${blaslapackLIBS} ${incLIBShdf} ${incLIBSfft} ${incLIBSmpi}")
set(FFLAGS "${OPT} ${LPATH} ${LIBS} ${IPATH} ${MODULE_FLAG} ${MODULE_PATH} ${FLG0}")
set(FFLAGS1 "${OPT} ${LPATH} ${LIBS} ${IPATH} ${MODULE_FLAG} ${MODULE_PATH} ${FLG1}")
set(FFLAGS_LOW "${OPTLOW} ${LPATH} ${LIBS} ${IPATH} ${MODULE_FLAG} ${MODULE_PATH} ${FLG0}")

# Create a minimalist Makefile.inc
file(WRITE ${CMAKE_SOURCE_DIR}/libolle/Makefile.inc
"# Generated by CMake - DO NOT EDIT
FC = ${CMAKE_Fortran_COMPILER}
CC = ${CMAKE_C_COMPILER}
CPP = ${CMAKE_CXX_COMPILER}

# Paths
OBJECT_PATH = ${OBJECT_PATH}
MODULE_PATH = ${MODULE_PATH}
DEST_PATH = ${DEST_PATH}

# Flags
MODULE_FLAG = ${MODULE_FLAG}
OPT = ${OPT}
OPTLOW = ${OPTLOW}
FLG0 = ${FLG0}
FLG1 = ${FLG1}

# Library paths
blaslapackLPATH = ${blaslapackLPATH}
blaslapackIPATH = ${blaslapackIPATH}
blaslapackLIBS = ${blaslapackLIBS}

incLPATHhdf = ${incLPATHhdf}
incIPATHhdf = ${incIPATHhdf}
incLIBShdf = ${incLIBShdf}

incLPATHfft = ${incLPATHfft}
incIPATHfft = ${incIPATHfft}
incLIBSfft = ${incLIBSfft}

incLPATHmpi = ${incLPATHmpi}
incIPATHmpi = ${incIPATHmpi}
incLIBSmpi = ${incLIBSmpi}

# Pre-combined flags as in original Makefile
LPATH = ${LPATH}
IPATH = ${IPATH} 
LIBS = ${LIBS}
FFLAGS = ${FFLAGS}
FFLAGS1 = ${FFLAGS1}
FFLAGS_LOW = ${FFLAGS_LOW}
")

# Add custom target to build using the original Makefile
add_custom_target(olle ALL
  COMMAND ${CMAKE_MAKE_PROGRAM} libolle.a
  WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/libolle
  COMMENT "Building libolle with original Makefile"
  VERBATIM
)
